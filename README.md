Magma is a minimal set of single-header modules for making games and apps in C/C++, inspired by libraries like [sokol](https://github.com/floooh/sokol) and [stb](https://github.com/nothings/stb).

## Headers

| Header                    | Description                                                |
| ------------------------- | ---------------------------------------------------------- |
| [`mgapp.h`](#mgapph)       | Platform layer — windowing, input, event loop             |
| [`mgfx.h`](#mgfxh)         | Render API abstraction — D3D11, OpenGL, OpenGL ES, Vulkan |
| [`mgmath.h`](#mgmathh)     | Math — vectors, matrices, quaternions                     |
| [`mgaudio.h`](#mgaudioh)   | Audio output — WASAPI, ALSA, Web Audio, WAV encoding/decoding |

## mgapp.h

A minimal platform abstraction supporting **Windows**, **Linux (Xlib)**, and **Web (Emscripten)**. macOS support is coming soon.

```c
#define MG_IMPL
#include "mgapp.h"

void on_start(void)
{

}

void on_end(void)
{

}

void on_update(void)
{

}

void on_event(const mgapp_event *event)
{

}

mgapp_init_info mgapp_main(void)
{
    return (mgapp_init_info) {
        .title = "My App",
        .events = {
            .start = on_start,
            .end = on_end,
            .update = on_update,
            .event = on_event
        }
    };
}
```

## mgfx.h

A render API abstraction supporting **DirectX 11**, **OpenGL**, **OpenGL ES 3**, **Vulkan**, and **Metal** (coming soon). Switch backends with a single `#define` before the include or in mgfx_init() during runtime.

```c
#define MG_IMPL
#include "mgapp.h"

#define MGFX_OPENGL
#include "mgfx.h"

#include "triangle.glsl.h" // generated by magma-shdc

static mgfx_pipeline pip;
static mgfx_buffer vb;

void on_start(void)
{
    mgfx_init(&(mgfx_init_info){
        .primary_handle = mgapp_primary_handle(),
        .secondary_handle = mgapp_secondary_handle(),
        .width = mgapp_width(),
        .height = mgapp_height(),
        .vsync = true
    });

    vb = mgfx_create_buffer(&(mgfx_buffer_create_info){
        .usage = MGFX_BUFFER_USAGE_VERTEX,
        .data = (float[]){
            0.0f, 0.5f, 1.0f, 0.5f, 0.5f,
            0.5f, -0.5f, 0.5f, 1.0f, 0.5f,
            -0.5f, -0.5f, 0.5f, 0.5f, 1.0f,
            0.0f, 0.5f, 1.0f, 0.5f, 0.5f
        },
        .size = 20 * sizeof(float)
    });

    pip = mgfx_create_pipeline(&(mgfx_pipeline_create_info){
        .vertex_layout = {
            MGFX_VERTEX_FORMAT_FLOAT2,
            MGFX_VERTEX_FORMAT_FLOAT3
        },
        .primitive_topology = MGFX_PRIMITIVE_TOPOLOGY_LINE_STRIP,
        .shader = get_triangle_shader(mgfx_get_shader_lang())
    });
}

void on_end(void)
{
    mgfx_destroy_buffer(vb);
    mgfx_destroy_pipeline(pip);
    mgfx_shutdown();
}

void on_update(void)
{
    if (mgfx_begin() != MGFX_RESULT_SUCCESS)
        return;

    mgfx_bind_pass(&(mgfx_pass_info){.clear = {0.05f, 0.05f, 0.05f, 1.0f}});
    mgfx_bind_pipeline(pip);
    mgfx_bind_vertex_buffer(vb);
    mgfx_draw(4, 0);

    mgfx_end();
}

void on_event(const mgapp_event *event)
{
    if (event->type == MGAPP_EVENT_RESIZE)
        mgfx_resize(event->window.width, event->window.height);
}

mgapp_init_info mgapp_main(void)
{
    return (mgapp_init_info) {
        .title = "Triangle",
        .events = {
            .start = on_start,
            .end = on_end,
            .update = on_update,
            .event = on_event
        }
    };
}
```

## mgmath.h

Vector, matrix and quaternion math. All functions are `static inline` — no translation unit needed.

```c
#include "mgmath.h"

// Vectors
mg_vec3 a = {1.0f, 0.0f, 0.0f};
mg_vec3 b = {0.0f, 1.0f, 0.0f};
mg_vec3 c = mg_vec3_cross(a, b);

// Quaternions
mg_vec4 q = mg_quat_from_euler(0.0f, MG_DEG2RAD * 45.0f, 0.0f);

// Matrices
mg_mat4 proj = mg_mat4_perspective(MG_DEG2RAD * 60.0f, 16.0f/9.0f, 0.1f, 1000.0f);
mg_mat4 view = mg_mat4_look_at((mg_vec3){0,2,5}, (mg_vec3){0,0,0}, (mg_vec3){0,1,0});
mg_mat4 mvp  = mg_mat4_mul(proj, view);
```

## Magma Shader Compiler (magma-shdc)

Magma ships its own shader cross-compiler built on [SPIRV-Cross](https://github.com/KhronosGroup/SPIRV-Cross) — similar in spirit to [sokol-shdc](https://github.com/floooh/sokol-tools). Write one GLSL source file with `@stage` annotations and compile it to a C header containing shader code for all target backends.

**Supported output languages:** `glsl430`, `glsl300es`, `hlsl5`, `msl`, `spirv`

### Writing a shader

```glsl
@name triangle
@stage vertex

in vec2 pos;
in vec3 color;
out vec3 frag_color;

void main()
{
    gl_Position = vec4(pos, 0.0, 1.0);
    frag_color = color;
}

@stage fragment

in vec3 frag_color;
out vec4 out_color;

void main()
{
    out_color = vec4(frag_color, 1.0);
}
```

### Compiling

```bash
magma-shdc --input triangle.glsl --output triangle.glsl.h --lang hlsl,glsl,spirv
```

This produces `triangle.glsl.h` which you `#include` directly. At runtime, pass the correct variant to your pipeline using `mgfx_get_shader_lang()`:

```c
.shader = get_triangle_shader(mgfx_get_shader_lang()),
```

```bat
./build-magma-shdc.bat
```

```bash
# Dependencies
sudo pacman -Sy clang base-devel spirv-cross vulkan-devel libegl xorg-server

# Build
./build-magma-shdc.sh
```

## mgaudio.h

A minimal audio output layer supporting **Windows** (WASAPI), **Linux** (ALSA, loaded dynamically via `dlopen` so there's no link-time dependency), and **Web** (Web Audio via Emscripten). Audio is delivered through a single callback that fills an interleaved `float` buffer. It also includes WAV decoding and encoding (PCM16, PCM24 and Float32).


```c
#define MG_IMPL
#include "mgapp.h"
#include "mgaudio.h"
#include "mgmath.h"

static float phase = 0.0f;

void audio_callback(float *output, uint32_t frames, void *user_data)
{
    const uint32_t channels = mgaudio_channels();
    const float step = 440.0f / (float)mgaudio_sample_rate();

    for (uint32_t i = 0; i < frames; i++)
    {
        const float sample = sinf(phase * 2.0f * 3.14159265f) * 0.2f;
        phase += step;
        if (phase >= 1.0f)
            phase -= 1.0f;

        for (uint32_t c = 0; c < channels; c++)
            output[i * channels + c] = sample;
    }
}

void on_start(void)
{
    mgaudio_init(&(mgaudio_init_info){
        .data_callback = audio_callback
    });
}

void on_end(void)
{
    mgaudio_shutdown();
}

mgapp_init_info mgapp_main(void)
{
    return (mgapp_init_info) {
        .title = "Audio",
        .events = {
            .start = on_start,
            .end = on_end
        }
    };
}
```

Playback can be paused and resumed with `mgaudio_stop()` and `mgaudio_start()`.

### WAV files

```c
mgaudio_wav wav;
if (mgaudio_wav_decode_file("sound.wav", &wav) == MGAUDIO_RESULT_SUCCESS)
{
    mgaudio_wav_encode_file("copy.wav", wav.samples, wav.frames, wav.channels, wav.sample_rate, MGAUDIO_WAV_FORMAT_PCM16);
    mgaudio_wav_free(&wav);
}
```
